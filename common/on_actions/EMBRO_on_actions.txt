		effect = {
			every_possible_country = { add_ideas = PNSD_hidden }
			every_possible_country = {
				add_timed_idea = {
					idea = EMBRO_startefectivnost
					days = 1
                }
			}
			MAL = {
				give_resource_rights = { receiver = ENG state = 336 }
				give_resource_rights = { receiver = ENG state = 333 }
			}
			INS = {
				give_resource_rights = { receiver = HOL state = 672 }
				give_resource_rights = { receiver = HOL state = 335 }
				give_resource_rights = { receiver = HOL state = 334 }
				give_resource_rights = { receiver = HOL state = 667 }
				give_resource_rights = { receiver = HOL state = 673 }
			}
			PHI = {
				give_resource_rights = { receiver = USA state = 623 }
				give_resource_rights = { receiver = USA state = 327 }
				give_resource_rights = { receiver = USA state = 628 }
				give_resource_rights = { receiver = USA state = 627 }
			}
			SOV = {
				give_resource_rights = { receiver = JAP state = 655 }
			}
			883 = { add_claim_by = EGY }
			551 = { add_claim_by = EGY }
			767 = { add_claim_by = EGY }
			886 = { add_claim_by = EGY }
			549 = { add_claim_by = EGY }
			887 = { add_claim_by = EGY }
			885 = { add_claim_by = EGY }
			884 = { add_claim_by = EGY }
		}
	}

	on_weekly = {
		effect = {
			if = {
				limit = {
					NOT = { has_idea = PNSD_hidden }
				}
				add_ideas = PNSD_hidden
			}
		}
	}

	on_puppet = {
		effect = {
			if = {
				limit = {
					NOT = { has_idea = PNSD_hidden }
				}
				add_ideas = PNSD_hidden
			}
		}
	}
	on_subject_free = {
		effect = {
			#make sure Dominions lose the dominion tag
			if = {
				limit = {
					original_tag = EGY
				}
				drop_cosmetic_tag = yes
			}
		}
	}
}
