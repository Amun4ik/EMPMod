country_event = {
	id = siam.1
	title = siam.1.t
	desc = siam.1.d
	picture = GFX_report_event_japanese_siam_politicians

	is_triggered_only = yes
	
	# Принять
	option = {
	    ai_chance = {
			factor = 50
			modifier = {
			    is_historical_focus_on = yes
				factor = 0
			}
		}
		name = siam.1.a
		SIA = {
            country_event = { id = siam.2 hours = 16 random_hours = 8 }
		}
		if = {
			limit = {
				ROOT = { owns_state = 1017 }
			}
		    SIA = {
			    transfer_state = 1017
            }
		}
		if = {
			limit = {
				ROOT = { owns_state = 1018 }
			}
	        SIA = {
			    transfer_state = 1018
            }
		}
	}
	# Отказать
	option = {
	    ai_chance = {
			factor = 50
		}
		name = siam.1.b
        custom_effect_tooltip = SIA_canleadtoborderwar
		SIA = {
            country_event = { id = siam.3 hours = 16 random_hours = 8 }
		}
	}
}
